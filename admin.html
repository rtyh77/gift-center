function fetchIPs() { const container = document.getElementById('ipContainer'); container.innerHTML = '
جاري تحديث البيانات...
'; db.collection("visitors").get() .then((snapshot) => { if (snapshot.empty) { container.innerHTML = '
لا توجد اتصالات مسجلة حالياً.
'; return; } container.innerHTML = ''; let docs = []; snapshot.forEach(doc => { docs.push({ id: doc.id, ...doc.data() }); }); // الأحدث أولاً docs.sort((a, b) => { let timeA = a.timestamp ? new Date(a.timestamp.toDate ? a.timestamp.toDate() : a.timestamp).getTime() : 0; let timeB = b.timestamp ? new Date(b.timestamp.toDate ? b.timestamp.toDate() : b.timestamp).getTime() : 0; return timeB - timeA; }); docs.forEach((data) => { // ========================= // البيانات الأساسية // ========================= const ip = data.ip || "غير معروف"; const country = translateText(data.country); const countryCode = data.country_code || "غير معروف"; const region = translateText(data.region); const city = translateText(data.city); const postal = data.postal || data.zip || "غير متوفر"; const isp = data.isp || data.organization || "غير معروف"; const organization = data.organization || data.isp || "غير معروف"; const asn = data.asn || "غير معروف"; const continent = data.continent || "غير معروف"; // ========================= // الموقع التقريبي من GeoIP // ========================= const latitude = data.lat !== undefined && data.lat !== null ? data.lat : "غير متوفر"; const longitude = data.lon !== undefined && data.lon !== null ? data.lon : "غير متوفر"; // ========================= // الوقت واللغة // ========================= const deviceTimezone = data.device_timezone || "غير معروف"; const utcOffset = data.utc_offset || "غير معروف"; const deviceLanguage = data.device_language || data.locale || "غير معروف"; // ========================= // الشبكة // ========================= const networkType = data.network_type || "غير معروف"; const effectiveType = data.effective_type || "غير متوفر"; const downlink = data.downlink !== undefined ? data.downlink + " Mbps" : "غير متوفر"; const rtt = data.rtt !== undefined ? data.rtt + " ms" : "غير متوفر"; const saveData = data.save_data === true ? "مفعّل" : data.save_data === false ? "غير مفعّل" : "غير معروف"; // ========================= // الجهاز // ========================= const deviceType = data.device_type || "غير معروف"; const os = data.os || "غير معروف"; const browser = data.browser || "غير معروف"; // ========================= // الشاشة // ========================= const screenWidth = data.screen_width || "غير معروف"; const screenHeight = data.screen_height || "غير معروف"; const viewportWidth = data.viewport_width || "غير معروف"; const viewportHeight = data.viewport_height || "غير معروف"; const dpr = data.device_pixel_ratio || "غير معروف"; const colorDepth = data.color_depth || "غير معروف"; const orientation = data.orientation || "غير معروف"; // ========================= // إمكانيات الجهاز // ========================= const cpuCores = data.hardware_concurrency || "غير معروف"; const deviceMemory = data.device_memory ? data.device_memory + " GB" : "غير متوفر"; const touchPoints = data.max_touch_points !== undefined ? data.max_touch_points : "غير معروف"; // ========================= // الخصوصية / المتصفح // ========================= const cookies = data.cookies_enabled === true ? "مفعّلة" : data.cookies_enabled === false ? "غير مفعّلة" : "غير معروف"; const dnt = data.do_not_track === "1" ? "مفعّل" : data.do_not_track === "0" ? "غير مفعّل" : "غير محدد"; const online = data.online === true ? "متصل" : data.online === false ? "غير متصل" : "غير معروف"; // ========================= // الصفحة والجلسة // ========================= const pageUrl = data.page_url || "غير متوفر"; const pagePath = data.page_path || "غير متوفر"; const pageTitle = data.page_title || "غير متوفر"; const referrer = data.referrer || "مباشر / غير متوفر"; const sessionId = data.session_id || "غير متوفر"; // ========================= // الوقت // ========================= const time = formatTime(data.timestamp); // ========================= // VPN / Proxy // ========================= const vpnDetected = data.vpn_detected === true || data.is_vpn === true; const isVpn = vpnDetected || isp.toLowerCase().includes("vpn") || isp.toLowerCase().includes("proxy"); // ========================= // العنصر // ========================= const item = document.createElement('div'); item.className = `ip-item ${isVpn ? 'vpn-warning' : ''}`; item.innerHTML = `
> IP: ${ip}
الموقع: ${country} | ${region} | ${city}
رمز الدولة: ${countryCode}
القارة: ${continent}
الرمز البريدي: ${postal}
الإحداثيات التقريبية: ${latitude}, ${longitude}
نوع الاتصال: ${networkType}
نوع الشبكة حسب المتصفح: ${effectiveType}
سرعة الاتصال التقريبية: ${downlink}
زمن الاستجابة RTT: ${rtt}
توفير البيانات: ${saveData}
مزود الخدمة: ${ isVpn ? `${isp}` : isp }
المنظمة: ${organization}
ASN: ${asn}
VPN / Proxy: ${ isVpn ? `مؤشر محتمل` : "لم يتم اكتشاف مؤشر" }
نوع الجهاز: ${deviceType}
نظام التشغيل: ${os}
المتصفح: ${browser}
لغة الجهاز: ${deviceLanguage}
المنطقة الزمنية: ${deviceTimezone}
UTC Offset: ${utcOffset}
دقة الشاشة: ${screenWidth} × ${screenHeight}
حجم نافذة الصفحة: ${viewportWidth} × ${viewportHeight}
Device Pixel Ratio: ${dpr}
عمق الألوان: ${colorDepth}
اتجاه الشاشة: ${orientation}
أنوية المعالج: ${cpuCores}
ذاكرة الجهاز: ${deviceMemory}
اللمس: ${touchPoints} نقطة
Cookies: ${cookies}
Do Not Track: ${dnt}
حالة الاتصال: ${online}
الصفحة الحالية: ${pageTitle}
المسار: ${pagePath}
المصدر السابق: ${referrer}
Session ID: ${sessionId}
TIME: ${time}
`; container.appendChild(item); }); }) .catch((error) => { container.innerHTML = `
خطأ في الجلب: ${error.message}
`; }); }
